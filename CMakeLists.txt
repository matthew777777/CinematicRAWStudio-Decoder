cmake_minimum_required(VERSION 3.20)
project(CinematicRAWStudioDecoder VERSION 0.1.0 LANGUAGES CXX)

set(CRAW_SPEC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../CinematicRAWStudio-Spec"
    CACHE PATH "Path to CinematicRAWStudio-Spec (golden vectors)")
option(CRAW_BUILD_TOOLS "Build craw_dec" ON)
option(CRAW_BUILD_MOUNT "Build the virtual CinemaDNG mount" ON)
option(CRAW_REQUIRE_MOUNT "Fail if mount dependencies are unavailable" OFF)
option(CRAW_BUILD_TESTS "Build conformance tests" ON)
option(CRAW_FUZZ "Build libFuzzer target (clang)" OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_library(CrawDecoder src/Decoder.cpp src/Reader.cpp)
add_library(CRAW::Decoder ALIAS CrawDecoder)
target_compile_features(CrawDecoder PUBLIC cxx_std_20)
target_include_directories(CrawDecoder PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(CrawDecoder PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(CrawDecoder PRIVATE /W4 /permissive-)
else()
    target_compile_options(CrawDecoder PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(CrawDecoder PUBLIC Threads::Threads)
if(ANDROID AND CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
    target_compile_options(CrawDecoder PRIVATE -march=armv8-a+crc)
endif()

if(CRAW_BUILD_TOOLS)
    add_executable(craw_dec tools/craw_dec.cpp)
    target_link_libraries(craw_dec PRIVATE CRAW::Decoder)

    # DaVinci Resolve workflow: CinemaDNG export and a virtual CinemaDNG mount.
    add_library(CrawClip STATIC tools/common/Clip.cpp)
    target_link_libraries(CrawClip PUBLIC CRAW::Decoder)
    add_executable(craw2dng tools/craw2dng.cpp)
    target_link_libraries(craw2dng PRIVATE CrawClip)

    install(TARGETS craw_dec craw2dng RUNTIME DESTINATION bin)
    if(CRAW_BUILD_MOUNT)
        set(_fuse_version 26)
        if(WIN32)
            set(_winfsp "$ENV{ProgramFiles\(x86\)}/WinFsp")
            find_path(FUSE_INCLUDE fuse.h PATHS "${_winfsp}/inc/fuse" "C:/Program Files (x86)/WinFsp/inc/fuse" NO_DEFAULT_PATH)
            if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
                set(_winfsp_name winfsp-a64)
            elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
                set(_winfsp_name winfsp-x64)
            else()
                set(_winfsp_name winfsp-x86)
            endif()
            find_library(FUSE_LIB NAMES ${_winfsp_name} PATHS "${_winfsp}/lib" "C:/Program Files (x86)/WinFsp/lib")
        elseif(APPLE)
            find_path(FUSE_INCLUDE fuse.h PATHS "/Library/Application Support/fuse-t/include/fuse" /usr/local/include/fuse NO_DEFAULT_PATH)
            find_library(FUSE_LIB NAMES fuse-t PATHS /usr/local/lib)
        else()
            find_package(PkgConfig QUIET)
            if(PKG_CONFIG_FOUND)
                pkg_check_modules(FUSE3 QUIET fuse3)
            endif()
            if(FUSE3_FOUND)
                set(FUSE_INCLUDE ${FUSE3_INCLUDE_DIRS})
                set(FUSE_LIB ${FUSE3_LINK_LIBRARIES})
                set(_fuse_version 31)
            else()
                find_path(FUSE_INCLUDE fuse.h PATHS /usr/include/fuse /usr/local/include/fuse)
                find_library(FUSE_LIB NAMES fuse)
            endif()
        endif()
        if(FUSE_INCLUDE AND FUSE_LIB)
            add_executable(craw_mount tools/craw_mount.cpp)
            target_include_directories(craw_mount PRIVATE ${FUSE_INCLUDE})
            target_compile_definitions(craw_mount PRIVATE FUSE_USE_VERSION=${_fuse_version} _FILE_OFFSET_BITS=64)
            target_link_libraries(craw_mount PRIVATE CrawClip ${FUSE_LIB})
            if(WIN32)
                target_compile_definitions(craw_mount PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
                if(MSVC)
                    target_link_options(craw_mount PRIVATE "/DELAYLOAD:${_winfsp_name}.dll")
                    target_link_libraries(craw_mount PRIVATE delayimp)
                endif()
            endif()
            install(TARGETS craw_mount RUNTIME DESTINATION bin)
        elseif(CRAW_REQUIRE_MOUNT)
            message(FATAL_ERROR "Mount support needs fuse-t on macOS, libfuse3-dev/fuse3-devel on Linux, or WinFsp with developer files on Windows. Install it or use --no-mount / -NoMount.")
        else()
            message(STATUS "craw_mount: filesystem driver not found, skipped")
        endif()
    endif()
endif()

if(CRAW_BUILD_TESTS)
    enable_testing()
    add_executable(craw_dec_conformance tests/ConformanceTests.cpp)
    target_link_libraries(craw_dec_conformance PRIVATE CRAW::Decoder)
    target_compile_definitions(craw_dec_conformance PRIVATE CRAW_VECTORS_DIR="${CRAW_SPEC_DIR}/vectors")
    add_test(NAME conformance COMMAND craw_dec_conformance)
    if(TARGET CrawClip)
        add_executable(craw_wav_tests tests/WavTests.cpp)
        target_include_directories(craw_wav_tests PRIVATE tools)
        target_link_libraries(craw_wav_tests PRIVATE CrawClip)
        add_test(NAME wav COMMAND craw_wav_tests)
    endif()
endif()

if(CRAW_FUZZ)
    add_executable(craw_fuzz_decoder tests/FuzzDecoder.cpp)
    target_compile_options(craw_fuzz_decoder PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(craw_fuzz_decoder PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(craw_fuzz_decoder PRIVATE CRAW::Decoder)
endif()

include(GNUInstallDirs)
install(TARGETS CrawDecoder EXPORT CrawDecoderTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CrawDecoderTargets NAMESPACE CRAW:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CrawDecoder)
